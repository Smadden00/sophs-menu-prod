Commands to create the database
CREATE TABLE reviews(review_id SERIAL UNIQUE NOT NULL PRIMARY KEY, rest_name TEXT NOT NULL, o_rating NUMERIC(3,1) NOT NULL CHECK (o_rating >= 0 AND o_rating <=10), price SMALLINT NOT NULL CHECK (price >= 1 AND price <= 4), taste NUMERIC(3,1) NOT NULL CHECK (taste >= 0 AND taste <=10), experience NUMERIC(3,1) NOT NULL CHECK (experience >= 0 AND experience <=10), description TEXT NOT NULL, user_encrypted VARCHAR(64) NOT NULL, city VARCHAR(50) NOT NULL, soph_submitted BOOLEAN NOT NULL, state_code CHAR(2) NOT NULL);

CREATE TABLE recipes(recipe_id SERIAL UNIQUE NOT NULL PRIMARY KEY, recipe_name TEXT NOT NULL, user_encrypted VARCHAR(64) NOT NULL, prep_time INTEGER NOT NULL CHECK (prep_time > 0), rating NUMERIC(3,1) NOT NULL CHECK (rating >= 0 AND rating <=10), meal TEXT, rec_img_url TEXT);

CREATE TABLE recipesComments(comment_id SERIAL UNIQUE NOT NULL PRIMARY KEY, recipe_id INT NOT NULL, comment VARCHAR(150) NOT NULL, user_encrypted VARCHAR(64) NOT NULL);
ALTER TABLE recipesComments ADD FOREIGN KEY (recipe_id) REFERENCES recipes(recipe_id) ON DELETE CASCADE;

CREATE TABLE recipe_instructions(instruction_id SERIAL UNIQUE NOT NULL PRIMARY KEY, recipe_id INTEGER NOT NULL, instruction_order INTEGER NOT NULL, instruction TEXT NOT NULL);
ALTER TABLE recipe_instructions ADD FOREIGN KEY (recipe_id) REFERENCES recipes(recipe_id) ON DELETE CASCADE;

CREATE TABLE recipe_ingredients(ingredient_id SERIAL UNIQUE NOT NULL PRIMARY KEY, recipe_id INTEGER NOT NULL, ingredient TEXT NOT NULL);
ALTER TABLE recipe_ingredients ADD FOREIGN KEY (recipe_id) REFERENCES recipes(recipe_id);

CREATE TABLE allergens(allergen_id SERIAL UNIQUE NOT NULL PRIMARY KEY, allergen TEXT NOT NULL UNIQUE);
CREATE TABLE allergen_recipe_ref(allergen_id INTEGER NOT NULL, recipe_id INTEGER NOT NULL);
ALTER TABLE allergen_recipe_ref ADD PRIMARY KEY (allergen_id, recipe_id);
ALTER TABLE allergen_recipe_ref ADD FOREIGN KEY (allergen_id) REFERENCES allergens(allergen_id);
ALTER TABLE allergen_recipe_ref ADD FOREIGN KEY (recipe_id) REFERENCES recipes(recipe_id);

CREATE TABLE cuisines(cuisine_id SERIAL UNIQUE NOT NULL PRIMARY KEY, cuisine TEXT NOT NULL UNIQUE);
CREATE TABLE cuisine_recipe_ref(cuisine_id INTEGER NOT NULL, recipe_id INTEGER NOT NULL);
ALTER TABLE cuisine_recipe_ref ADD PRIMARY KEY (cuisine_id, recipe_id);
ALTER TABLE cuisine_recipe_ref ADD FOREIGN KEY (cuisine_id) REFERENCES cuisines(cuisine_id) ON DELETE CASCADE;
ALTER TABLE cuisine_recipe_ref ADD FOREIGN KEY (recipe_id) REFERENCES recipes(recipe_id) ON DELETE CASCADE;

CREATE TABLE rest_types(rest_type_id SERIAL UNIQUE NOT NULL PRIMARY KEY, rest_type TEXT NOT NULL UNIQUE);
CREATE TABLE rest_type_review_ref(rest_type_id INTEGER NOT NULL, review_id INTEGER NOT NULL);
ALTER TABLE rest_type_review_ref ADD PRIMARY KEY (rest_type_id, review_id);
ALTER TABLE rest_type_review_ref ADD FOREIGN KEY (rest_type_id) REFERENCES rest_types(rest_type_id) ON DELETE CASCADE;
ALTER TABLE rest_type_review_ref ADD FOREIGN KEY (review_id) REFERENCES reviews(review_id) ON DELETE CASCADE;
